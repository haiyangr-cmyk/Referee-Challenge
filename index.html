<script>


// 示例关卡（来自文档）

const levels = [

{

    q:"C罗手臂自然张开碰到球，进球是否有效？",

    options:["有效","无效"],

    answer:0,

    explain:"手臂自然位置，不算手球"

},

{

    q:"德国球员越位半个肩，进球有效吗？",

    options:["有效","无效"],

    answer:0,

    explain:"整体越位才算"

},

{

    q:"飞铲踢膝盖，应如何判罚？",

    options:["红牌","黄牌"],

    answer:0,

    explain:"危险动作直接红牌"

}

];


let current = 0;

let score = 0;

let mode = "good";


function startGame(m){

    mode = m;

    document.getElementById("start").classList.add("hidden");

    document.getElementById("game").classList.remove("hidden");

    showQuestion();

}


function showQuestion(){

    let level = levels[current];

    document.getElementById("progress").innerText = 

        "第 " + (current+1) + " 关";

    document.getElementById("question").innerText = level.q;


    let html = "";

    level.options.forEach((opt,i)=>{

        html += `<div class="button" onclick="choose(${i})">${opt}</div>`;

    });

    document.getElementById("answers").innerHTML = html;

}


function choose(i){

    let level = levels[current];


    document.getElementById("game").classList.add("hidden");

    document.getElementById("result").classList.remove("hidden");


    if(mode==="good"){

        if(i===level.answer){

            score++;

            document.getElementById("resultText").innerText="✅ 判罚正确";

        }else{

            document.getElementById("resultText").innerText="❌ 判罚错误";

        }

        document.getElementById("explain").innerText=level.explain;


    }else{

        document.getElementById("resultText").innerText="😈 越黑越强！";

        document.getElementById("explain").innerText="黑哨评分+1";

    }

}


function next(){

    current++;

    if(current>=levels.length){

        alert("游戏结束！得分："+score);

        location.reload();

        return;

    }

    document.getElementById("result").classList.add("hidden");

    document.getElementById("game").classList.remove("hidden");

    showQuestion();

}


</script>
